81: def parse_ingredient(text):
82:     text = text.strip()
83:     m_mixed = re.match(r'^(\d+)\s+([½⅓⅔¼¾⅛⅜⅝⅞])\s+(.*)$', text)
84:     if m_mixed:
85:         whole = int(m_mixed.group(1))
86:         frac = FRACTIONS[m_mixed.group(2)]
87:         val = whole + frac
88:         display = f"{m_mixed.group(1)} {m_mixed.group(2)}"
89:         return val, display, m_mixed.group(3)
90: 
91:     m_frac = re.match(r'^([½⅓⅔¼¾⅛⅜⅝⅞])\s+(.*)$', text)
92:     if m_frac:
93:         frac_char = m_frac.group(1)
94:         return FRACTIONS[frac_char], frac_char, m_frac.group(2)
95: 
96:     m_num = re.match(r'^(\d+(?:\.\d+)?)\s+(.*)$', text)
97:     if m_num:
98:         val = float(m_num.group(1))
99:         display = m_num.group(1)
100:         return val, display, m_num.group(2)
101: 
102:     return None, None, text
